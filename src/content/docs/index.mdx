## Why Attest?

**57% of organizations run AI agents in production without systematic testing.** Most testing tools force expensive LLM-as-judge calls for every assertion. Attest takes a different approach: an 8-layer graduated assertion pipeline that starts with free deterministic checks and only escalates to costly LLM evaluation when cheaper layers can't answer the question.

## What It Looks Like

### Python

```python
from attest import agent, expect

@agent("weather-agent")
def weather_agent(builder, query: str) -> dict:
    # Your agent implementation: record tool calls on the builder
    builder.add_tool_call("get_weather", args={"city": "Tokyo"})
    ...

def test_weather_agent(attest):
    result = weather_agent(query="What's the weather in Tokyo?")

    chain = (
        expect(result)
        .required_tools(["get_weather"])                          # Layer 3: Free
        .output_contains("Tokyo")                                 # Layer 4: Free
        .output_similar_to("Current weather conditions in Tokyo")  # Layer 5: ≪$0.001
        .passes_judge("Provides accurate, helpful weather information")  # Layer 6: ~$0.001
    )
    attest.evaluate(chain)
```

### TypeScript

```typescript
import { agent, attestExpect } from '@attest-ai/core';
import { evaluate } from '@attest-ai/vitest';

const weatherAgent = agent('weather-agent', (builder, args) => {
  // Your agent implementation: record tool calls on the builder
  builder.addToolCall('get_weather', { args: { city: 'Tokyo' } });
  return { message: '...' };
});

test('weather agent returns weather data', async () => {
  const result = weatherAgent({ query: 'What is the weather in Tokyo?' });

  const chain = attestExpect(result)
    .requiredTools(['get_weather'])                               // Layer 3: Free
    .outputContains('Tokyo')                                      // Layer 4: Free
    .outputSimilarTo('Current weather conditions in Tokyo')       // Layer 5: ≪$0.001
    .passesJudge('Provides accurate, helpful weather information'); // Layer 6: ~$0.001

  await evaluate(chain);
});
```
